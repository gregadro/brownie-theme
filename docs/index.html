/**
 * A tiny bakery queue.
 * @see https://github.com/gregadro/brownie-theme
 */
export enum Topping {
  SaltedCaramel = "salted-caramel",
  Walnut = "walnut",
  None = "none",
}

export interface Brownie {
  readonly id: number;
  topping: Topping;
  bakedAt?: Date;
}

const MAX_TRAY_SIZE = 16;
const slugPattern = /^[a-z]+(-[a-z]+)*$/g;

export class Oven<T extends Brownie = Brownie> {
  private tray: T[] = [];

  constructor(private readonly temperature: number = 180) {}

  add(item: T): this {
    if (this.tray.length >= MAX_TRAY_SIZE) {
      throw new Error(`Tray is full (${MAX_TRAY_SIZE} pieces)\n`);
    }
    this.tray.push(item);
    return this;
  }

  async bake(minutes = 25): Promise<T[]> {
    await new Promise((resolve) => setTimeout(resolve, minutes * 1000));
    const now = new Date();
    return this.tray.map((b) => ({ ...b, bakedAt: now }));
  }
}

const oven = new Oven(175);
const isSlug = slugPattern.test(Topping.SaltedCaramel);

oven
  .add({ id: 1, topping: Topping.SaltedCaramel })
  .add({ id: 2, topping: isSlug ? Topping.Walnut : Topping.None })
  .bake()
  .then((tray) => console.log(tray.length, "brownies ready", true, null));

bakery.pyrecipe.json
"""Bakery inventory, a small Python example for the Brownie theme."""

from __future__ import annotations

import re
from dataclasses import dataclass, field
from enum import Enum
from typing import Iterator

LOW_STOCK = 3
SKU_PATTERN = re.compile(r"^BRW-\d{4}$")


class Flavor(Enum):
    SALTED_CARAMEL = "salted caramel"
    WALNUT = "walnut"
    DOUBLE_CHOCOLATE = "double chocolate"


@dataclass(frozen=True)
class Brownie:
    sku: str
    flavor: Flavor
    price: float = 2.5
    tags: tuple[str, ...] = field(default_factory=tuple)

    def __post_init__(self) -> None:
        if not SKU_PATTERN.match(self.sku):
            raise ValueError(f"Invalid SKU: {self.sku!r}\n")


class Inventory:
    def __init__(self) -> None:
        self._stock: dict[str, int] = {}

    def add(self, brownie: Brownie, quantity: int = 1) -> Inventory:
        self._stock[brownie.sku] = self._stock.get(brownie.sku, 0) + quantity
        return self

    def low_stock(self) -> Iterator[str]:
        yield from (sku for sku, qty in self._stock.items() if qty <= LOW_STOCK)

    @property
    def total(self) -> int:
        return sum(self._stock.values())


if __name__ == "__main__":
    caramel = Brownie("BRW-0001", Flavor.SALTED_CARAMEL, tags=("bestseller",))
    walnut = Brownie("BRW-0002", Flavor.WALNUT, price=2.9)

    inventory = Inventory().add(caramel, 12).add(walnut, 2)
    # Only walnut is running low
    print(list(inventory.low_stock()), inventory.total, None, True)

# Salted Caramel Brownies

> The best brownie is the one you are eating right now.

## Ingredients

- **12 tablespoons** unsalted butter, _melted_
- 1 cup granulated sugar
- ~~raisins~~ (never)
- 1 teaspoon `salt`

## Steps

1. Preheat the oven to **180 °C**.
2. Mix everything, see [the recipe][tasty].
3. Bake for _25 minutes_.

```js
const brownie = bake({ minutes: 25 });
```

| Ingredient | Amount |
| ---------- | -----: |
| Butter     |  170 g |
| Sugar      |  200 g |

From the kitchen

Why a brownie

I made Brownie in 2022 for my own Vue work. The idea was simple: writing code should feel as good as eating a salted caramel brownie. Warm, soft on the eyes, a little sweet, with a pinch of salt.

This year it got a proper re-bake. The colors are clearer and easier to tell apart, there is a light Blondie for sunny days, and two new flavors if you like your brownie with berries or cherries.

If it makes your editor a little cozier, that is all I wanted.

Grega

Install

Into the oven

  1. Install the extension

    Search for Brownie in the Extensions view (Ctrl Shift X), or get it from the Marketplace.

  2. Choose a flavor

    Open the theme picker with Ctrl K Ctrl T (⌘ K ⌘ T on a Mac) and type Brownie or Blondie.

  3. Let the sun decide

    Brownie at night, Blondie by day. Add this to your settings.json:

    {
  "window.autoDetectColorScheme": true,
  "workbench.preferredDarkColorTheme": "Brownie",
  "workbench.preferredLightColorTheme": "Blondie"
}
